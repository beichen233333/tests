cmake_minimum_required(VERSION 3.18)
project(AstraeOfflineBackend VERSION 1.0.0 LANGUAGES C CXX)
option(ASTRAE_BUILD_TESTS "Build host-side regression tests" ON)
set(ASTRAE_ORIGINAL_SO "" CACHE FILEPATH "Optional original AArch64 SO for hook manifest differential checks")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(ASTRAE_SOURCES src/json.cpp src/backend.cpp src/il2cpp_runtime.cpp src/managed_wire.cpp src/hook_runtime.cpp src/api.cpp compat/src/cJSON_compat.cpp compat/src/store_core.cpp compat/src/original_store_abi.cpp compat/src/errno_compat.cpp)
find_package(Threads REQUIRED)
add_library(astraeoffline_objects OBJECT ${ASTRAE_SOURCES})
target_include_directories(astraeoffline_objects PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include ${CMAKE_CURRENT_SOURCE_DIR}/compat/include PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_definitions(astraeoffline_objects PRIVATE ASTRAE_BACKEND_BUILD=1)
set_target_properties(astraeoffline_objects PROPERTIES POSITION_INDEPENDENT_CODE YES)
if(NOT ANDROID)
  target_compile_options(astraeoffline_objects PRIVATE -Wall -Wextra -Wpedantic -Wno-misleading-indentation)
endif()
# Compatibility artifact: only the original library's public C/JNI ABI is visible.
add_library(astraeoffline SHARED $<TARGET_OBJECTS:astraeoffline_objects>)
target_link_libraries(astraeoffline PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
target_link_options(astraeoffline PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/compat/original_exports.map")
target_include_directories(astraeoffline PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/compat/include)
set_target_properties(astraeoffline PROPERTIES OUTPUT_NAME astraeoffline)
if(ANDROID)
  # Preserve the original library's public platform dependency set. --no-as-needed
  # keeps liblog.so in DT_NEEDED even when the compatibility stage does not yet
  # exercise every original logging path; libc/libdl remain platform dependencies.
  target_link_options(astraeoffline PRIVATE "LINKER:--no-as-needed" "LINKER:-z,max-page-size=16384")
  target_link_libraries(astraeoffline PRIVATE log)
else()
  set_target_properties(astraeoffline PROPERTIES BUILD_RPATH "$ORIGIN")
  # Developer/test artifact keeps the reconstruction control API visible; it is not the replacement library.
  add_library(astraeoffline_dev SHARED $<TARGET_OBJECTS:astraeoffline_objects>)
  target_link_libraries(astraeoffline_dev PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
  target_include_directories(astraeoffline_dev PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include ${CMAKE_CURRENT_SOURCE_DIR}/compat/include PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
  set_target_properties(astraeoffline_dev PROPERTIES OUTPUT_NAME astraeoffline_dev BUILD_RPATH "$ORIGIN")
endif()
